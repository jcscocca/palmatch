> **For agentic workers:** REQUIRED SUB-SKILL: Use superpowers:subagent-driven-development (recommended) or superpowers:executing-plans to implement this plan task-by-task. Steps use checkbox (`- [ ]`) syntax for tracking.

**Goal:** Static Palworld breeding workbench (parents→child, target→parents, chains, mutations, passive odds) with real game sprites, per spec `docs/superpowers/specs/2026-07-26-palmatch-design.md`.

**Architecture:** Vite + React + TS SPA. Build-time pipeline transforms palcalc's datamined JSON (pinned commit) into compact artifacts; pure-TS engine consumes them; Zustand store syncs three slots + tab to URL hash; UI is a single Workbench screen.

**Tech Stack:** Vite 8, React 19, TypeScript, Zustand, vitest, tsx (script runner), sharp (sprite conversion). No component library. No backend.

**Pinned upstream:** `tylercamp/palcalc @ be2ec7a95c521dea6591469c051e7cb0f6658065` (2026-07-25).

## Artifact formats (contract between pipeline / engine / UI)

- `public/data/pals.json`: `{ version: {palcalcCommit, gameVersion, refreshedAt}, pals: PalRecord[] }`, sorted by `InternalName`, **array index = pal index everywhere**. `PalRecord = { id: string /*InternalName*/, name: string, dex: string /*"143" or "143B" when IsVariant*/, types: string[] /*palpedia strings*/, power: number, priority: number, maleProb: number /*0..1*/, guaranteed: string[] /*passive InternalNames*/, sprite: string /*"/sprites/{id}.webp"*/ }`

```ts
export const GENDER_SENTINEL = 0xffff;
export function breed(ds: Dataset, a: number, b: number): BreedResult[] {
  const cell = ds.matrix[a * ds.pals.length + b];
  if (cell === GENDER_SENTINEL) {
    return ds.combos.genderLocked
      .filter(g => (g.a === a && g.b === b) || (g.a === b && g.b === a))
      .map(g => ({ child: g.child, condition: g.aGender === 'F' ? 'gender-AF' as const : 'gender-AM' as const,
                   lock: { aId: ds.pals[g.a].id, aGender: g.aGender, bId: ds.pals[g.b].id, bGender: g.bGender } }));
    // NOTE: branch on g.aGender, NOT g.a === a — both entries share the same a index, so
    // the positional check mislabels both results. lock carries render info for the UI.
  }
  const isUnique = ds.combos.unique.some(u => (u.a === a && u.b === b) || (u.a === b && u.b === a));
  return [{ child: cell, condition: a === b ? 'same-species' : isUnique ? 'unique' : 'standard' }];
}
export function findParents(ds: Dataset, target: number): ParentCombo[] {
  const n = ds.pals.length, out: ParentCombo[] = [];
  for (let a = 0; a < n; a++)
    for (let b = a; b < n; b++)
      if (ds.matrix[a * n + b] === target) out.push({ a, b, badges: badgesFor(ds, a, b) });
  for (const g of ds.combos.genderLocked) if (g.child === target) out.push({ a: Math.min(g.a,g.b), b: Math.max(g.a,g.b), badges: ['gender-locked'] });
  return out;
}
```

```ts
export interface ChainStep { a: number; b: number; child: number }
export function findChains(ds: Dataset, starters: number[], target: number, maxDepth = 6): ChainStep[] | null {
  const n = ds.pals.length;
  const cost = new Int8Array(n).fill(127); const via = new Int32Array(n).fill(-1); // via = a*n+b
  for (const s of starters) cost[s] = 0;
  for (let round = 0; round < maxDepth; round++) {
    let changed = false;
    const known: number[] = []; for (let i = 0; i < n; i++) if (cost[i] <= round) known.push(i);
    for (const a of known) for (const b of known) { if (b < a) continue;
      const child = ds.matrix[a * n + b]; if (child === GENDER_SENTINEL) continue;
      const c = Math.max(cost[a], cost[b]) + 1; // parents must exist before breeding; see note
      const alt = cost[a] + cost[b] + 1;        // total breeds (greedy, shared subtrees may double-count)
      if (alt < cost[child] && alt <= maxDepth) { cost[child] = alt as any; via[child] = a * n + b; changed = true; }
    }
    if (!changed) break;
  }
  if (cost[target] > maxDepth) return null;
  const steps: ChainStep[] = []; const emit = (p: number) => { if (via[p] < 0) return;
    const a = Math.floor(via[p] / n), b = via[p] % n; emit(a); emit(b);
    if (!steps.some(s => s.child === p)) steps.push({ a, b, child: p }); };
  emit(target); return steps;
}
```

```ts
const r5 = (x: number) => Math.round(2 * x + 0.5) >> 1;
export function mutationPool(ds: Dataset, a: number, b: number): { child: number; weight: number }[] {
  const { rankCoefficient, rankDiffPenalty, randomCoefficient } = ds.mutation;
  const ra = ds.pals[a].power, rb = ds.pals[b].power, lo = Math.min(ra, rb);
  const h = r5(lo * rankCoefficient) + r5(Math.abs(ra - rb) * rankDiffPenalty);
  const p = Math.max(1, r5(lo * randomCoefficient));
  const eligible = eligibleByRank(ds); // memoized: all pal indices sorted by power
  const counts = new Map<number, number>();
  for (let r = Math.max(1, h + 1); r <= h + p; r++) { const c = nearestByRank(eligible, ds, r); counts.set(c, (counts.get(c) ?? 0) + 1); }
  const total = [...counts.values()].reduce((s, x) => s + x, 0);
  return [...counts].map(([child, k]) => ({ child, weight: k / total })).sort((x, y) => y.weight - x.weight);
}
```

```ts
const K_WEIGHTS = [0.4, 0.3, 0.2, 0.1]; // P(inherit k = 1..4) from parents' pool
function choose(n: number, k: number): number { if (k < 0 || k > n) return 0; let r = 1; for (let i = 0; i < k; i++) r = (r * (n - i)) / (i + 1); return r; }
/** P(child directly inherits ALL of `desired` (subset of parents' union of size n)). Estimate; random-fill slots ignored. */
export function passiveOdds(poolSize: number, desiredCount: number): number {
  if (desiredCount === 0) return 1; if (desiredCount > Math.min(poolSize, 4)) return 0;
  let p = 0;
  for (let k = desiredCount; k <= Math.min(4, poolSize); k++)
    p += K_WEIGHTS[k - 1] * (choose(poolSize - desiredCount, k - desiredCount) / choose(poolSize, k));
  return p;
}
```
